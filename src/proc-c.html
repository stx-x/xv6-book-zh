struct {
  struct spinlock lock;
  struct proc proc[NPROC];
} ptable;

static struct proc *initproc;

int nextpid = 1;
extern void forkret(void);
extern void trapret(void);

static void wakeup1(void *chan);

void
pinit(void)
{
  initlock(&ptable.lock, "ptable");
}

// 必须在关闭中断的情况下调用
int
cpuid() {
  return mycpu()-cpus;
}

// 必须在关闭中断的情况下调用，以免调用者在
// 读取 lapicid 与遍历循环之间被重新调度。
struct cpu*
mycpu(void)
{
  int apicid, i;
  
  if(readeflags()&FL_IF)
    panic("mycpu called with interrupts enabled\n");
  
  apicid = lapicid();
  // APIC ID 不保证连续。也许我们应当维护一张
  // 反向映射表，或者保留一个寄存器来存放 &cpus[i]。
  for (i = 0; i < ncpu; ++i) {
    if (cpus[i].apicid == apicid)
      return &cpus[i];
  }
  panic("unknown apicid\n");
}

// 关闭中断，以免在从 cpu 结构读取 proc 的
// 过程中被重新调度
struct proc*
myproc(void) {
  struct cpu *c;
  struct proc *p;
  pushcli();
  c = mycpu();
  p = c->proc;
  popcli();
  return p;
}

// PAGEBREAK: 32
// 在进程表中查找状态为 UNUSED 的 proc。
// 若找到，把状态改为 EMBRYO，并初始化
// 在内核中运行所需的状态。
// 否则返回 0。
static struct proc*
allocproc(void)
{
  struct proc *p;
  char *sp;

  acquire(&ptable.lock);

  for(p = ptable.proc; p < &ptable.proc[NPROC]; p++)
    if(p->state == UNUSED)
      goto found;

  release(&ptable.lock);
  return 0;

found:
  p->state = EMBRYO;
  p->pid = nextpid++;

  release(&ptable.lock);

  // 分配内核栈。
  if((p->kstack = kalloc()) == 0){
    p->state = UNUSED;
    return 0;
  }
  sp = p->kstack + KSTACKSIZE;

  // 为陷阱帧留出空间。
  sp -= sizeof *p->tf;
  p->tf = (struct trapframe*)sp;

  // 设置新上下文，使其从 forkret 开始执行，
  // 而 forkret 会返回到 trapret。
  sp -= 4;
  *(uint*)sp = (uint)trapret;

  sp -= sizeof *p->context;
  p->context = (struct context*)sp;
  memset(p->context, 0, sizeof *p->context);
  p->context->eip = (uint)forkret;

  return p;
}

// PAGEBREAK: 32
// 设置第一个用户进程。
void
userinit(void)
{
  struct proc *p;
  extern char _binary_initcode_start[], _binary_initcode_size[];

  p = allocproc();
  
  initproc = p;
  if((p->pgdir = setupkvm()) == 0)
    panic("userinit: out of memory?");
  inituvm(p->pgdir, _binary_initcode_start, (int)_binary_initcode_size);
  p->sz = PGSIZE;
  memset(p->tf, 0, sizeof(*p->tf));
  p->tf->cs = (SEG_UCODE << 3) | DPL_USER;
  p->tf->ds = (SEG_UDATA << 3) | DPL_USER;
  p->tf->es = p->tf->ds;
  p->tf->ss = p->tf->ds;
  p->tf->eflags = FL_IF;
  p->tf->esp = PGSIZE;
  p->tf->eip = 0;  // initcode.S 的起始处

  safestrcpy(p->name, "initcode", sizeof(p->name));
  p->cwd = namei("/");

  // 对 p->state 的这次赋值让其他核
  // 可以运行该进程。acquire 强制上述
  // 写入可见，而且这里也需要这把锁，
  // 因为该赋值可能不是原子的。
  acquire(&ptable.lock);

  p->state = RUNNABLE;

  release(&ptable.lock);
}

// 把当前进程的内存增长 n 字节。
// 成功返回 0，失败返回 -1。
int
growproc(int n)
{
  uint sz;
  struct proc *curproc = myproc();

  sz = curproc->sz;
  if(n > 0){
    if((sz = allocuvm(curproc->pgdir, sz, sz + n)) == 0)
      return -1;
  } else if(n < 0){
    if((sz = deallocuvm(curproc->pgdir, sz, sz + n)) == 0)
      return -1;
  }
  curproc->sz = sz;
  switchuvm(curproc);
  return 0;
}

// 创建一个新进程，以 p 为父进程。
// 设置栈，使其像从系统调用返回一样返回。
// 调用者必须把返回的 proc 状态设为 RUNNABLE。
int
fork(void)
{
  int i, pid;
  struct proc *np;
  struct proc *curproc = myproc();

  // 分配进程。
  if((np = allocproc()) == 0){
    return -1;
  }

  // 从 proc 复制进程状态。
  if((np->pgdir = copyuvm(curproc->pgdir, curproc->sz)) == 0){
    kfree(np->kstack);
    np->kstack = 0;
    np->state = UNUSED;
    return -1;
  }
  np->sz = curproc->sz;
  np->parent = curproc;
  *np->tf = *curproc->tf;

  // 清除 %eax，使 fork 在子进程中返回 0。
  np->tf->eax = 0;

  for(i = 0; i < NOFILE; i++)
    if(curproc->ofile[i])
      np->ofile[i] = filedup(curproc->ofile[i]);
  np->cwd = idup(curproc->cwd);

  safestrcpy(np->name, curproc->name, sizeof(curproc->name));

  pid = np->pid;

  acquire(&ptable.lock);

  np->state = RUNNABLE;

  release(&ptable.lock);

  return pid;
}

// 退出当前进程。不返回。
// 已退出的进程停留在僵尸状态，
// 直到其父进程调用 wait() 得知它已退出。
void
exit(void)
{
  struct proc *curproc = myproc();
  struct proc *p;
  int fd;

  if(curproc == initproc)
    panic("init exiting");

  // 关闭所有打开的文件。
  for(fd = 0; fd < NOFILE; fd++){
    if(curproc->ofile[fd]){
      fileclose(curproc->ofile[fd]);
      curproc->ofile[fd] = 0;
    }
  }

  begin_op();
  iput(curproc->cwd);
  end_op();
  curproc->cwd = 0;

  acquire(&ptable.lock);

  // 父进程可能正睡眠在 wait() 中。
  wakeup1(curproc->parent);

  // 把被遗弃的子进程交给 init。
  for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
    if(p->parent == curproc){
      p->parent = initproc;
      if(p->state == ZOMBIE)
        wakeup1(initproc);
    }
  }

  // 跳入调度器，永不返回。
  curproc->state = ZOMBIE;
  sched();
  panic("zombie exit");
}

// 等待一个子进程退出并返回其 pid。
// 若该进程没有子进程则返回 -1。
int
wait(void)
{
  struct proc *p;
  int havekids, pid;
  struct proc *curproc = myproc();
  
  acquire(&ptable.lock);
  for(;;){
    // 扫描进程表，寻找已退出的子进程。
    havekids = 0;
    for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
      if(p->parent != curproc)
        continue;
      havekids = 1;
      if(p->state == ZOMBIE){
        // 找到一个。
        pid = p->pid;
        kfree(p->kstack);
        p->kstack = 0;
        freevm(p->pgdir);
        p->pid = 0;
        p->parent = 0;
        p->name[0] = 0;
        p->killed = 0;
        p->state = UNUSED;
        release(&ptable.lock);
        return pid;
      }
    }

    // 如果没有子进程，等待没有意义。
    if(!havekids || curproc->killed){
      release(&ptable.lock);
      return -1;
    }

    // 等待子进程退出。（见 proc_exit 中对 wakeup1 的调用。）
    sleep(curproc, &ptable.lock);  // DOC: wait-sleep
  }
}

// PAGEBREAK: 42
// 每 CPU 的进程调度器。
// 每个 CPU 在完成自身设置后调用 scheduler()。
// 调度器永不返回。它循环执行：
// - 选择一个要运行的进程
// - swtch 到该进程开始运行
// - 最终该进程通过 swtch 把控制权
// 交还给调度器。
void
scheduler(void)
{
  struct proc *p;
  struct cpu *c = mycpu();
  c->proc = 0;
  
  for(;;){
    // 在此处理器上启用中断。
    sti();

    // 遍历进程表，寻找要运行的进程。
    acquire(&ptable.lock);
    for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
      if(p->state != RUNNABLE)
        continue;

      // 切换到选中的进程。释放 ptable.lock、
      // 再在跳回我们之前重新获取它，
      // 是那个进程自己的职责。
      c->proc = p;
      switchuvm(p);
      p->state = RUNNING;

      swtch(&(c->scheduler), p->context);
      switchkvm();

      // 该进程暂时运行结束。
      // 它应当在返回之前已经改变了自己的 p->state。
      c->proc = 0;
    }
    release(&ptable.lock);

  }
}

// 进入调度器。必须只持有 ptable.lock
// 并且已经改变了 proc->state。保存并恢复
// intena，因为 intena 是这个内核线程的属性，
// 而不是这个 CPU 的属性。它本应是
// proc->intena 与 proc->ncli，但那样会在
// 少数“持有锁但没有进程”的地方
// 出问题。
void
sched(void)
{
  int intena;
  struct proc *p = myproc();

  if(!holding(&ptable.lock))
    panic("sched ptable.lock");
  if(mycpu()->ncli != 1)
    panic("sched locks");
  if(p->state == RUNNING)
    panic("sched running");
  if(readeflags()&FL_IF)
    panic("sched interruptible");
  intena = mycpu()->intena;
  swtch(&p->context, mycpu()->scheduler);
  mycpu()->intena = intena;
}

// 放弃 CPU 一个调度轮次。
void
yield(void)
{
  acquire(&ptable.lock);  // DOC: yieldlock
  myproc()->state = RUNNABLE;
  sched();
  release(&ptable.lock);
}

// fork 出来的子进程第一次被 scheduler() 调度时
// 会 swtch 到这里。“返回”用户空间。
void
forkret(void)
{
  static int first = 1;
  // 仍然持有来自 scheduler 的 ptable.lock。
  release(&ptable.lock);

  if (first) {
    // 有些初始化函数必须在常规进程的上下文中
    // 运行（例如它们会调用 sleep），因此不能
    // 从 main() 中运行。
    first = 0;
    iinit(ROOTDEV);
    initlog(ROOTDEV);
  }

  // 返回到“调用者”，实际上是 trapret（见 allocproc）。
}

// 原子地释放锁并在 chan 上睡眠。
// 被唤醒时重新获取该锁。
void
sleep(void *chan, struct spinlock *lk)
{
  struct proc *p = myproc();
  
  if(p == 0)
    panic("sleep");

  if(lk == 0)
    panic("sleep without lk");

  // 必须先获取 ptable.lock，才能
  // 改变 p->state 然后调用 sched。
  // 一旦持有 ptable.lock，我们就能
  // 保证不会错过任何唤醒
  // （wakeup 在持有 ptable.lock 的情况下运行），
  // 因此释放 lk 是安全的。
  if(lk != &ptable.lock){  // DOC: sleeplock0
    acquire(&ptable.lock);  // DOC: sleeplock1
    release(lk);
  }
  // 进入睡眠。
  p->chan = chan;
  p->state = SLEEPING;

  sched();

  // 整理收尾。
  p->chan = 0;

  // 重新获取原来的锁。
  if(lk != &ptable.lock){  // DOC: sleeplock2
    release(&ptable.lock);
    acquire(lk);
  }
}

// PAGEBREAK!
// 唤醒所有在 chan 上睡眠的进程。
// 必须持有 ptable 锁。
static void
wakeup1(void *chan)
{
  struct proc *p;

  for(p = ptable.proc; p < &ptable.proc[NPROC]; p++)
    if(p->state == SLEEPING && p->chan == chan)
      p->state = RUNNABLE;
}

// 唤醒所有在 chan 上睡眠的进程。
void
wakeup(void *chan)
{
  acquire(&ptable.lock);
  wakeup1(chan);
  release(&ptable.lock);
}

// 杀死给定 pid 的进程。
// 进程在返回用户空间之前不会退出
// （见 trap.c 中的 trap）。
int
kill(int pid)
{
  struct proc *p;

  acquire(&ptable.lock);
  for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
    if(p->pid == pid){
      p->killed = 1;
      // 必要时把进程从睡眠中唤醒。
      if(p->state == SLEEPING)
        p->state = RUNNABLE;
      release(&ptable.lock);
      return 0;
    }
  }
  release(&ptable.lock);
  return -1;
}

// PAGEBREAK: 36
// 把进程列表打印到控制台。用于调试。
// 当用户在控制台输入 ^P 时运行。
// 不加锁，以免让已经卡住的机器更糟。
void
procdump(void)
{
  static char *states[] = {
  [UNUSED]    "unused",
  [EMBRYO]    "embryo",
  [SLEEPING]  "sleep ",
  [RUNNABLE]  "runble",
  [RUNNING]   "run   ",
  [ZOMBIE]    "zombie"
  };
  int i;
  struct proc *p;
  char *state;
  uint pc[10];

  for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
    if(p->state == UNUSED)
      continue;
    if(p->state >= 0 && p->state < NELEM(states) && states[p->state])
      state = states[p->state];
    else
      state = "???";
    cprintf("%d %s %s", p->pid, state, p->name);
    if(p->state == SLEEPING){
      getcallerpcs((uint*)p->context->ebp+2, pc);
      for(i=0; i<10 && pc[i] != 0; i++)
        cprintf(" %p", pc[i]);
    }
    cprintf("\n");
  }
}
