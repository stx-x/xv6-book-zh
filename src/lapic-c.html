// 见 Intel 处理器手册第三卷第 8 章与附录 C。

// 本地 APIC 寄存器，除以 4 后用作 uint[] 的下标。
#define ID      (0x0020/4)   // ID
#define VER     (0x0030/4)   // 版本
#define TPR     (0x0080/4)   // 任务优先级
#define EOI     (0x00B0/4)   // EOI
#define SVR     (0x00F0/4)   // 伪中断向量
  #define ENABLE     0x00000100   // 单元使能
#define ESR     (0x0280/4)   // 错误状态
#define ICRLO   (0x0300/4)   // 中断命令
  #define INIT       0x00000500   // INIT/RESET
  #define STARTUP    0x00000600   // 启动 IPI
  #define DELIVS     0x00001000   // 投递状态
  #define ASSERT     0x00004000   // 断言中断（相对于撤销断言）
  #define DEASSERT   0x00000000
  #define LEVEL      0x00008000   // 电平触发
  #define BCAST      0x00080000   // 发送给所有 APIC，包括自身。
  #define BUSY       0x00001000
  #define FIXED      0x00000000
#define ICRHI   (0x0310/4)   // 中断命令 [63:32]
#define TIMER   (0x0320/4)   // 本地向量表 0（定时器）
  #define X1         0x0000000B   // 计数不分频
  #define PERIODIC   0x00020000   // 周期模式
#define PCINT   (0x0340/4)   // 性能计数器 LVT
#define LINT0   (0x0350/4)   // 本地向量表 1（LINT0）
#define LINT1   (0x0360/4)   // 本地向量表 2（LINT1）
#define ERROR   (0x0370/4)   // 本地向量表 3（ERROR）
  #define MASKED     0x00010000   // 中断被屏蔽
#define TICR    (0x0380/4)   // 定时器初始计数
#define TCCR    (0x0390/4)   // 定时器当前计数
#define TDCR    (0x03E0/4)   // 定时器分频配置

volatile uint *lapic;  // 在 mp.c 中初始化

// PAGEBREAK!
static void
lapicw(int index, int value)
{
  lapic[index] = value;
  lapic[ID];  // 通过读取来等待写入完成
}

void
lapicinit(void)
{
  if(!lapic)
    return;

  // 启用本地 APIC；设置伪中断向量。
  lapicw(SVR, ENABLE | (T_IRQ0 + IRQ_SPURIOUS));

  // 定时器以总线频率反复从 lapic[TICR] 递减计数，
  // 计到零后发出一次中断。
  // 如果 xv6 更在意精确计时，
  // TICR 会用外部时间源来校准。
  lapicw(TDCR, X1);
  lapicw(TIMER, PERIODIC | (T_IRQ0 + IRQ_TIMER));
  lapicw(TICR, 10000000);

  // 禁用逻辑中断线。
  lapicw(LINT0, MASKED);
  lapicw(LINT1, MASKED);

  // 在提供该中断入口的机器上
  // 禁用性能计数器溢出中断。
  if(((lapic[VER]>>16) & 0xFF) >= 4)
    lapicw(PCINT, MASKED);

  // 把错误中断映射到 IRQ_ERROR。
  lapicw(ERROR, T_IRQ0 + IRQ_ERROR);

  // 清除错误状态寄存器（需要连续两次写入）。
  lapicw(ESR, 0);
  lapicw(ESR, 0);

  // 应答所有未处理的中断。
  lapicw(EOI, 0);

  // 发送 Init Level De-Assert 以同步仲裁 ID。
  lapicw(ICRHI, 0);
  lapicw(ICRLO, BCAST | INIT | LEVEL);
  while(lapic[ICRLO] & DELIVS)
    ;

  // 在 APIC 上启用中断（但不在处理器上）。
  lapicw(TPR, 0);
}

int
lapicid(void)
{
  if (!lapic)
    return 0;
  return lapic[ID] >> 24;
}

// 应答中断。
void
lapiceoi(void)
{
  if(lapic)
    lapicw(EOI, 0);
}

#define CMOS_PORT    0x70
#define CMOS_RETURN  0x71

// 让额外的处理器从 addr 处的入口代码开始运行。
// 见《多处理器规范》附录 B。
void
lapicstartap(uchar apicid, uint addr)
{
  int i;
  ushort *wrv;

  // “BSP 必须把 CMOS 关闭码初始化为 0AH，
  // 并把热复位向量（位于 40:67 的 DWORD）指向
  // AP 启动代码，然后才能执行［通用启动算法］。”
  outb(CMOS_PORT, 0xF);  // 偏移 0xF 是关闭码
  outb(CMOS_PORT+1, 0x0A);
  wrv = (ushort*)P2V((0x40<<4 | 0x67));  // 热复位向量
  wrv[0] = 0;
  wrv[1] = addr >> 4;

  // “通用启动算法。”
  // 发送 INIT（电平触发）中断以复位其他 CPU。
  lapicw(ICRHI, apicid<<24);
  lapicw(ICRLO, INIT | LEVEL | ASSERT);
  microdelay(200);
  lapicw(ICRLO, INIT | LEVEL);
  microdelay(100);    // 本应是 10ms，但在 Bochs 里太慢！

  // 发送启动 IPI（两次！）以进入代码。
  // 常规硬件只在因 INIT 而停机的状态下
  // 才接受 STARTUP。因此第二次
  // 本应被忽略，但它是 Intel 官方算法的一部分。
  // Bochs 会对第二次报怨。那只能怪 Bochs。
  for(i = 0; i < 2; i++){
    lapicw(ICRHI, apicid<<24);
    lapicw(ICRLO, STARTUP | (addr>>12));
    microdelay(200);
  }
}

#define CMOS_STATA   0x0a
#define CMOS_STATB   0x0b
#define CMOS_UIP    (1 << 7)        // RTC 更新进行中

#define SECS    0x00
#define MINS    0x02
#define HOURS   0x04
#define DAY     0x07
#define MONTH   0x08
#define YEAR    0x09

static uint
cmos_read(uint reg)
{
  outb(CMOS_PORT,  reg);
  microdelay(200);

  return inb(CMOS_RETURN);
}

static void
fill_rtcdate(struct rtcdate *r)
{
  r->second = cmos_read(SECS);
  r->minute = cmos_read(MINS);
  r->hour   = cmos_read(HOURS);
  r->day    = cmos_read(DAY);
  r->month  = cmos_read(MONTH);
  r->year   = cmos_read(YEAR);
}

// qemu 似乎使用 24 小时制 GWT，且数值为 BCD 编码
void
cmostime(struct rtcdate *r)
{
  struct rtcdate t1, t2;
  int sb, bcd;

  sb = cmos_read(CMOS_STATB);

  bcd = (sb & (1 << 2)) == 0;

  // 确保 CMOS 在我们读取期间不修改时间
  for(;;) {
    fill_rtcdate(&t1);
    if(cmos_read(CMOS_STATA) & CMOS_UIP)
        continue;
    fill_rtcdate(&t2);
    if(memcmp(&t1, &t2, sizeof(t1)) == 0)
      break;
  }

  // 转换
  if(bcd) {
#define    CONV(x)     (t1.x = ((t1.x >> 4) * 10) + (t1.x & 0xf))
    CONV(second);
    CONV(minute);
    CONV(hour  );
    CONV(day   );
    CONV(month );
    CONV(year  );
#undef     CONV
  }

  *r = t1;
  r->year += 2000;
}
