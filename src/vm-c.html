pde_t *kpgdir;  // 供 scheduler() 使用

// 设置 CPU 的内核段描述符。
// 在每个 CPU 进入时运行一次。
void
seginit(void)
{
  struct cpu *c;

  // 用恒等映射把“逻辑”地址映射到虚拟地址。
  // 内核与用户不能共用同一个 CODE 描述符，
  // 因为它必须具有 DPL_USR，而 CPU 禁止
  // 从 CPL=0 发生到 DPL=3 的中断。
  c = &cpus[cpuid()];
  c->gdt[SEG_KCODE] = SEG(STA_X|STA_R, 0, 0xffffffff, 0);
  c->gdt[SEG_KDATA] = SEG(STA_W, 0, 0xffffffff, 0);
  c->gdt[SEG_UCODE] = SEG(STA_X|STA_R, 0, 0xffffffff, DPL_USER);
  c->gdt[SEG_UDATA] = SEG(STA_W, 0, 0xffffffff, DPL_USER);
  lgdt(c->gdt, sizeof(c->gdt));
}

// 返回页表 pgdir 中与虚拟地址 va 对应的
// PTE 的地址。若 alloc!=0，
// 则创建所需的页表页。
static pte_t *
walkpgdir(pde_t *pgdir, const void *va, int alloc)
{
  pde_t *pde;
  pte_t *pgtab;

  pde = &pgdir[PDX(va)];
  if(*pde & PTE_P){
    pgtab = (pte_t*)P2V(PTE_ADDR(*pde));
  } else {
    if(!alloc || (pgtab = (pte_t*)kalloc()) == 0)
      return 0;
    // 确保所有这些 PTE_P 位都为零。
    memset(pgtab, 0, PGSIZE);
    // 这里的权限过于宽松，但必要时可以
    // 由页表项中的权限
    // 进一步加以限制。
    *pde = V2P(pgtab) | PTE_P | PTE_W | PTE_U;
  }
  return &pgtab[PTX(va)];
}

// 为从 va 开始的虚拟地址创建 PTE，指向
// 从 pa 开始的物理地址。va 与 size 可能
// 不是页对齐的。
static int
mappages(pde_t *pgdir, void *va, uint size, uint pa, int perm)
{
  char *a, *last;
  pte_t *pte;

  a = (char*)PGROUNDDOWN((uint)va);
  last = (char*)PGROUNDDOWN(((uint)va) + size - 1);
  for(;;){
    if((pte = walkpgdir(pgdir, a, 1)) == 0)
      return -1;
    if(*pte & PTE_P)
      panic("remap");
    *pte = pa | perm | PTE_P;
    if(a == last)
      break;
    a += PGSIZE;
    pa += PGSIZE;
  }
  return 0;
}

// 每个进程一张页表，另有一张在
// CPU 不运行任何进程时使用（kpgdir）。内核在系统调用与中断期间
// 使用当前进程的页表；
// 页保护位阻止用户代码使用内核的
// 映射。
//
// setupkvm() 与 exec() 为每张页表建立如下映射：
//
// 0..KERNBASE：用户内存（text+data+stack+heap），映射到
// 内核分配的物理内存
// KERNBASE..KERNBASE+EXTMEM：映射到 0..EXTMEM（用于 I/O 空间）
// KERNBASE+EXTMEM..data：映射到 EXTMEM..V2P(data)，
// 用于内核的指令与只读数据
// data..KERNBASE+PHYSTOP：映射到 V2P(data)..PHYSTOP，
// 即可读写数据与空闲物理内存
// 0xfe000000..0：直接映射（ioapic 等设备）
//
// 内核在 V2P(end) 与物理内存末尾（PHYSTOP）之间
// 为其堆和用户内存分配物理内存
// （可直接从 end..P2V(PHYSTOP) 寻址）。

// 这张表定义内核的映射，它们出现在
// 每个进程的页表中。
static struct kmap {
  void *virt;
  uint phys_start;
  uint phys_end;
  int perm;
} kmap[] = {
 { (void*)KERNBASE, 0,             EXTMEM,    PTE_W}, // I/O 空间
 { (void*)KERNLINK, V2P(KERNLINK), V2P(data), 0},     // 内核代码 + 只读数据
 { (void*)data,     V2P(data),     PHYSTOP,   PTE_W}, // 内核数据 + 内存
 { (void*)DEVSPACE, DEVSPACE,      0,         PTE_W}, // 更多设备
};

// 建立页表的内核部分。
pde_t*
setupkvm(void)
{
  pde_t *pgdir;
  struct kmap *k;

  if((pgdir = (pde_t*)kalloc()) == 0)
    return 0;
  memset(pgdir, 0, PGSIZE);
  if (P2V(PHYSTOP) > (void*)DEVSPACE)
    panic("PHYSTOP too high");
  for(k = kmap; k < &kmap[NELEM(kmap)]; k++)
    if(mappages(pgdir, k->virt, k->phys_end - k->phys_start,
                (uint)k->phys_start, k->perm) < 0) {
      freevm(pgdir);
      return 0;
    }
  return pgdir;
}

// 为调度器进程分配一张用于内核地址空间的
// 页表。
void
kvmalloc(void)
{
  kpgdir = setupkvm();
  switchkvm();
}

// 把硬件页表寄存器切换到仅含内核的页表，
// 供没有进程运行时使用。
void
switchkvm(void)
{
  lcr3(V2P(kpgdir));   // 切换到内核页表
}

// 把 TSS 与硬件页表切换到进程 p 对应的内容。
void
switchuvm(struct proc *p)
{
  if(p == 0)
    panic("switchuvm: no process");
  if(p->kstack == 0)
    panic("switchuvm: no kstack");
  if(p->pgdir == 0)
    panic("switchuvm: no pgdir");

  pushcli();
  mycpu()->gdt[SEG_TSS] = SEG16(STS_T32A, &mycpu()->ts,
                                sizeof(mycpu()->ts)-1, 0);
  mycpu()->gdt[SEG_TSS].s = 0;
  mycpu()->ts.ss0 = SEG_KDATA << 3;
  mycpu()->ts.esp0 = (uint)p->kstack + KSTACKSIZE;
  // 在 eflags 中设置 IOPL=0，并让 iomb 超出 tss 段限长，
  // 从而禁止用户空间执行 I/O 指令（例如 inb 与 outb）
  mycpu()->ts.iomb = (ushort) 0xFFFF;
  ltr(SEG_TSS << 3);
  lcr3(V2P(p->pgdir));  // 切换到进程的地址空间
  popcli();
}

// 把 initcode 装入 pgdir 的地址 0 处。
// sz 必须小于一页。
void
inituvm(pde_t *pgdir, char *init, uint sz)
{
  char *mem;

  if(sz >= PGSIZE)
    panic("inituvm: more than a page");
  mem = kalloc();
  memset(mem, 0, PGSIZE);
  mappages(pgdir, 0, PGSIZE, V2P(mem), PTE_W|PTE_U);
  memmove(mem, init, sz);
}

// 把一个程序段装入 pgdir。addr 必须页对齐，
// 且从 addr 到 addr+sz 的页必须已经映射。
int
loaduvm(pde_t *pgdir, char *addr, struct inode *ip, uint offset, uint sz)
{
  uint i, pa, n;
  pte_t *pte;

  if((uint) addr % PGSIZE != 0)
    panic("loaduvm: addr must be page aligned");
  for(i = 0; i < sz; i += PGSIZE){
    if((pte = walkpgdir(pgdir, addr+i, 0)) == 0)
      panic("loaduvm: address should exist");
    pa = PTE_ADDR(*pte);
    if(sz - i < PGSIZE)
      n = sz - i;
    else
      n = PGSIZE;
    if(readi(ip, P2V(pa), offset+i, n) != n)
      return -1;
  }
  return 0;
}

// 分配页表与物理内存，把进程从 oldsz 增长到
// newsz，两者不必页对齐。返回新的大小，出错返回 0。
int
allocuvm(pde_t *pgdir, uint oldsz, uint newsz)
{
  char *mem;
  uint a;

  if(newsz >= KERNBASE)
    return 0;
  if(newsz < oldsz)
    return oldsz;

  a = PGROUNDUP(oldsz);
  for(; a < newsz; a += PGSIZE){
    mem = kalloc();
    if(mem == 0){
      cprintf("allocuvm out of memory\n");
      deallocuvm(pgdir, newsz, oldsz);
      return 0;
    }
    memset(mem, 0, PGSIZE);
    if(mappages(pgdir, (char*)a, PGSIZE, V2P(mem), PTE_W|PTE_U) < 0){
      cprintf("allocuvm out of memory (2)\n");
      deallocuvm(pgdir, newsz, oldsz);
      kfree(mem);
      return 0;
    }
  }
  return newsz;
}

// 释放用户页，把进程大小从 oldsz 收缩到
// newsz。oldsz 与 newsz 不必页对齐，newsz 也
// 不必小于 oldsz。oldsz 可以大于进程的实际
// 大小。返回新的进程大小。
int
deallocuvm(pde_t *pgdir, uint oldsz, uint newsz)
{
  pte_t *pte;
  uint a, pa;

  if(newsz >= oldsz)
    return oldsz;

  a = PGROUNDUP(newsz);
  for(; a  < oldsz; a += PGSIZE){
    pte = walkpgdir(pgdir, (char*)a, 0);
    if(!pte)
      a = PGADDR(PDX(a) + 1, 0, 0) - PGSIZE;
    else if((*pte & PTE_P) != 0){
      pa = PTE_ADDR(*pte);
      if(pa == 0)
        panic("kfree");
      char *v = P2V(pa);
      kfree(v);
      *pte = 0;
    }
  }
  return newsz;
}

// 释放一张页表以及用户部分的
// 所有物理内存页。
void
freevm(pde_t *pgdir)
{
  uint i;

  if(pgdir == 0)
    panic("freevm: no pgdir");
  deallocuvm(pgdir, KERNBASE, 0);
  for(i = 0; i < NPDENTRIES; i++){
    if(pgdir[i] & PTE_P){
      char * v = P2V(PTE_ADDR(pgdir[i]));
      kfree(v);
    }
  }
  kfree((char*)pgdir);
}

// 清除某一页上的 PTE_U。用于在用户栈
// 下方创建一个不可访问的页。
void
clearpteu(pde_t *pgdir, char *uva)
{
  pte_t *pte;

  pte = walkpgdir(pgdir, uva, 0);
  if(pte == 0)
    panic("clearpteu");
  *pte &= ~PTE_U;
}

// 给定父进程的页表，为子进程
// 创建一份副本。
pde_t*
copyuvm(pde_t *pgdir, uint sz)
{
  pde_t *d;
  pte_t *pte;
  uint pa, i, flags;
  char *mem;

  if((d = setupkvm()) == 0)
    return 0;
  for(i = 0; i < sz; i += PGSIZE){
    if((pte = walkpgdir(pgdir, (void *) i, 0)) == 0)
      panic("copyuvm: pte should exist");
    if(!(*pte & PTE_P))
      panic("copyuvm: page not present");
    pa = PTE_ADDR(*pte);
    flags = PTE_FLAGS(*pte);
    if((mem = kalloc()) == 0)
      goto bad;
    memmove(mem, (char*)P2V(pa), PGSIZE);
    if(mappages(d, (void*)i, PGSIZE, V2P(mem), flags) < 0) {
      kfree(mem);
      goto bad;
    }
  }
  return d;

bad:
  freevm(d);
  return 0;
}

// PAGEBREAK!
// 把用户虚拟地址映射到内核地址。
char*
uva2ka(pde_t *pgdir, char *uva)
{
  pte_t *pte;

  pte = walkpgdir(pgdir, uva, 0);
  if((*pte & PTE_P) == 0)
    return 0;
  if((*pte & PTE_U) == 0)
    return 0;
  return (char*)P2V(PTE_ADDR(*pte));
}

// 把 p 处的 len 字节复制到页表 pgdir 中的用户地址 va。
// 在 pgdir 不是当前页表时最有用。
// uva2ka 保证这只对 PTE_U 页有效。
int
copyout(pde_t *pgdir, uint va, void *p, uint len)
{
  char *buf, *pa0;
  uint n, va0;

  buf = (char*)p;
  while(len > 0){
    va0 = (uint)PGROUNDDOWN(va);
    pa0 = uva2ka(pgdir, (char*)va0);
    if(pa0 == 0)
      return -1;
    n = PGSIZE - (va - va0);
    if(n > len)
      n = len;
    memmove(pa0 + (va - va0), buf, n);
    len -= n;
    buf += n;
    va = va0 + PGSIZE;
  }
  return 0;
}
