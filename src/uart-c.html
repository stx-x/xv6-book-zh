// Intel 8250 串口（UART）。

#include "types.h"
#include "defs.h"
#include "param.h"
#include "traps.h"
#include "spinlock.h"
#include "sleeplock.h"
#include "fs.h"
#include "file.h"
#include "mmu.h"
#include "proc.h"
#include "x86.h"

#define COM1    0x3f8

static int uart;    // 是否存在 uart？

void
uartinit(void)
{
  char *p;

  // 关闭 FIFO
  outb(COM1+2, 0);

  // 9600 波特率，8 位数据，1 位停止位，无校验。
  outb(COM1+3, 0x80);    // 解锁除数锁存器
  outb(COM1+0, 115200/9600);
  outb(COM1+1, 0);
  outb(COM1+3, 0x03);    // 锁定除数锁存器，8 位数据。
  outb(COM1+4, 0);
  outb(COM1+1, 0x01);    // 启用接收中断。

  // 若状态为 0xFF，则没有串口。
  if(inb(COM1+5) == 0xFF)
    return;
  uart = 1;

  // 应答先前存在的中断条件；
  // 启用中断。
  inb(COM1+2);
  inb(COM1+0);
  ioapicenable(IRQ_COM1, 0);

  // 宣告我们已就位。
  for(p="xv6...\n"; *p; p++)
    uartputc(*p);
}

void
uartputc(int c)
{
  int i;

  if(!uart)
    return;
  for(i = 0; i < 128 && !(inb(COM1+5) & 0x20); i++)
    microdelay(10);
  outb(COM1+0, c);
}

static int
uartgetc(void)
{
  if(!uart)
    return -1;
  if(!(inb(COM1+5) & 0x01))
    return -1;
  return inb(COM1+0);
}

void
uartintr(void)
{
  consoleintr(uartgetc);
}
