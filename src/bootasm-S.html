#include "asm.h"
#include "memlayout.h"
#include "mmu.h"

# 启动第一个 CPU：切换到 32 位保护模式，跳入 C 代码。
# BIOS 把这段代码从硬盘第一个扇区装入
# 物理地址 0x7c00 处的内存，并在实模式下开始执行，
# 此时 %cs=0、%ip=7c00。

.code16                       # 按 16 位模式汇编
.globl start
start:
  cli                         # BIOS 已启用中断；将其禁用

  # 把数据段寄存器 DS、ES、SS 清零。
  xorw    %ax,%ax             # 把 %ax 置零
  movw    %ax,%ds             # -> 数据段
  movw    %ax,%es             # -> 附加段
  movw    %ax,%ss             # -> 栈段

  # 物理地址线 A20 被固定为零，以便最初那些
  # 只有 2 MB 内存的 PC 能运行假定内存为 1 MB 的软件。现在取消这一限制。
seta20.1:
  inb     $0x64,%al               # 等待不忙
  testb   $0x2,%al
  jnz     seta20.1

  movb    $0xd1,%al               # 0xd1 -> 端口 0x64
  outb    %al,$0x64

seta20.2:
  inb     $0x64,%al               # 等待不忙
  testb   $0x2,%al
  jnz     seta20.2

  movb    $0xdf,%al               # 0xdf -> 端口 0x60
  outb    %al,$0x60

  # 从实模式切换到保护模式。使用一个引导 GDT，使
  # 虚拟地址直接映射到物理地址，从而在切换过程中
  # 有效的内存映射保持不变。
  lgdt    gdtdesc
  movl    %cr0, %eax
  orl     $CR0_PE, %eax
  movl    %eax, %cr0

// PAGEBREAK!
  # 通过一次远跳转完成向 32 位保护模式的切换，
  # 以重新加载 %cs 与 %eip。段描述符被设置为不做
  # 转换，因此映射仍是恒等映射。
  ljmp    $(SEG_KCODE<<3), $start32

.code32  # 告诉汇编器现在开始生成 32 位代码。
start32:
  # 设置保护模式下的数据段寄存器
  movw    $(SEG_KDATA<<3), %ax    # 我们的数据段选择子
  movw    %ax, %ds                # -> DS：数据段
  movw    %ax, %es                # -> ES：附加段
  movw    %ax, %ss                # -> SS：栈段
  movw    $0, %ax                 # 把尚未使用的段清零
  movw    %ax, %fs                # -> FS
  movw    %ax, %gs                # -> GS

  # 设置栈指针并调用 C 代码。
  movl    $start, %esp
  call    bootmain

  # 如果 bootmain 返回（它不该返回），在 Bochs 下运行时
  # 触发一个 Bochs 断点，然后循环。
  movw    $0x8a00, %ax            # 0x8a00 -> 端口 0x8a00
  movw    %ax, %dx
  outw    %ax, %dx
  movw    $0x8ae0, %ax            # 0x8ae0 -> 端口 0x8a00
  outw    %ax, %dx
spin:
  jmp     spin

# 引导 GDT
.p2align 2                                # 强制 4 字节对齐
gdt:
  SEG_NULLASM                             # 空段
  SEG_ASM(STA_X|STA_R, 0x0, 0xffffffff)   # 代码段
  SEG_ASM(STA_W, 0x0, 0xffffffff)         # 数据段

gdtdesc:
  .word   (gdtdesc - gdt - 1)             # sizeof(gdt) - 1
  .long   gdt                             # gdt 的地址
